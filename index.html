07b · Integration Sample

How subscriber segments sync to Flodesk

Sincronización de segmentos de suscriptores a Flodesk

Action Scheduler worker that syncs WooCommerce customer segments to Flodesk in batches of 50.

wc-flodesk-pro/flodeskintegration.php 
public function worker_custom_segment_batch( $segment_id, $customers ) {
    if ( ! $this->is_license_active() ) {
        $this->log( 'worker_custom_segment_batch: licencia inactiva. Segmento ' . $segment_id );
        return;
    }
    if ( empty( $customers ) || ! is_array( $customers ) ) {
        $this->log( 'worker_custom_segment_batch: customers vacíos. Segmento ' . $segment_id );
        return;
    }

    $subscribers = array();
    foreach ( $customers as $customer ) {
        if ( empty( $customer['email'] ) ) {
            continue;
        }
        $subscribers[] = array(
            'email'       => $customer['email'],
            'first_name'  => isset( $customer['first_name'] ) ? $customer['first_name'] : '',
            'segment_ids' => array( (string) $segment_id ),
        );
    }

    $this->log( 'worker_custom_segment_batch: segmento ' . $segment_id . ' - ' . count( $subscribers ) . ' suscriptores a enviar.' );

    if ( empty( $subscribers ) ) {
        return;
    }

    $response = $this->flodesk_batch_upsert_subscribers( $subscribers );
    if ( is_wp_error( $response ) ) {
        $this->log( 'worker_custom_segment_batch: WP_Error segmento ' . $segment_id . ' - ' . $response->get_error_message() );
        return;
    }
    $code = (int) wp_remote_retrieve_response_code( $response );
    $body = wp_remote_retrieve_body( $response );
    $this->log( 'worker_custom_segment_batch: segmento ' . $segment_id . ' - respuesta HTTP ' . $code . ' body: ' . substr( $body, 0, 200 ) );
    $this->log_if_failed( $response, sprintf( 'custom segment batch -> segmento %s (%d suscriptores)', $segment_id, count( $subscribers ) ) );
}
